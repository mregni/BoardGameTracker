import { Card, CardGrid } from '@astrojs/starlight/components';

## Features

<CardGrid>
	<Card title="Dashboard">
		Total games, players and sessions, collection value, play time and recent
		activity on one page.
	</Card>
	<Card title="Game library">
		Cover art, search and filters, ownership state, expansions and the price
		paid per game.
	</Card>
	<Card title="Session logging">
		Log plays with players, scores, location, duration and a comment, including
		co-op games with a team result.
	</Card>
	<Card title="Player statistics">
		Win rates, favourite games, play time and badges per player.
	</Card>
	<Card title="Leaderboard">
		Ranks every player by wins, win rate and plays, with podiums and time
		played.
	</Card>
	<Card title="Player vs player">
		Head-to-head records, win rates and shared play history for two players.
	</Card>
	<Card title="BoardGameGeek import">
		Import a single game by its BGG id, or a whole BGG collection by username.
	</Card>
	<Card title="Price tracking">
		Current shop price and stock of a game through a changedetection.io
		instance.
	</Card>
	<Card title="Rules assistant">
		Ask questions about uploaded PDF rulebooks, answered by a local AI model
		with page citations.
	</Card>
	<Card title="Game night planner">
		Plan game nights, invite players and collect their RSVPs through a link
		that works without an account.
	</Card>
	<Card title="Shelf of Shame">
		Lists the owned games that have not been played for a configurable number
		of months.
	</Card>
	<Card title="Loans">
		Tracks which game is lent to whom and when it is due back.
	</Card>
	<Card title="Locations">
		Where you play, with the sessions, games and players per place.
	</Card>
	<Card title="Accounts and single sign-on">
		Local accounts with Admin, User and Reader roles, plus one OpenID Connect
		provider (Keycloak, Authentik, ...).
	</Card>
</CardGrid>
